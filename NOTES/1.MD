## showing a image

```py

import cv2

IMAGE = "assets\img.jpg"

def main():
    
    print("Hello from opencv-lrn!")

    image = cv2.imread(IMAGE)

    if image is None:
        print("Error: check path")
    
    if image is not None:
        cv2.imshow("IMAGE", image)      # shows image 
        cv2.waitKey(0)                  # checks for window close btn
        cv2.destroyAllWindows()         # closes all privious cv window
        
    else:
        print("success: 🤖🎊🎉")

```
