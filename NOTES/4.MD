## ROTATION

### SYNTAX

```PY

M = cv2.getRotationMatrix2D(center=, angle=, scale=)

rotate = cv2.warpAffine(image, M , (w, h))
        

```

### code

```PY


IMAGE = "assets/img.jpg"

def main():
    
    print("Hello from opencv-lrn!")

    image = cv2.imread(IMAGE)


    if image is None:
        print("err: !")
    
    else:
        (h, w) = image.shape[:2]
        
        center = (w //2, h // 2)
        
        M = cv2.getRotationMatrix2D(center, 90, 1.0)
        rotate = cv2.warpAffine(image, M , (w, h))
        
        cv2.imshow("Image", image)
        cv2.imshow("Rotated", rotate)
        cv2.imwrite("assets/rotate.jpg", rotate)
        cv2.waitKey(0)
        
```

## image flipping

### sybntax

```py

fliped = cv2.flip(image, flip)

                        flip = [
                            1 = horizontal
                            0 = vertical
                            -1 = both
                        ]

```

### Code

```py

IMAGE = "assets/img.jpg"

def main():
    
    print("Hello from opencv-lrn!")

    image = cv2.imread(IMAGE)


    if image is None:
        print("err: !")
    
    else:

        flopwe_h = cv2.flip(image, 1)
        flopwe_v = cv2.flip(image, 0)
        flopwe_both= cv2.flip(image, -1)
        
        cv2.imshow("Image", image)
        cv2.imshow("flopwe_h", flopwe_h)
        cv2.imshow("flopwe_v", flopwe_v)
        cv2.imshow("flopwe_both", flopwe_both)
        cv2.waitKey(0)
        
        




```
