```PY

import cv2

IMAGE = "assets/img.jpg"

def main():
    
    print("Hello from opencv-lrn!")

    image = cv2.imread(IMAGE)


    if image is None:
        print("err: !")
    
    else:

        pn1 = (50, 100)
        pn2 = (300, 10)
        color = (0, 255, 21)
        thickness = 4
        
        cv2.line(image, pn1, pn2, color, thickness)
        
        cv2.imshow("image", image)
        cv2.waitKey(0)
        cv2.destroyAllWindows()

        


```
